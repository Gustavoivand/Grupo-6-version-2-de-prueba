\subsection{UC01: Registrar Usuario}

\textbf{Actores:} Usuario no registrado

\textbf{Precondiciones:} El usuario no tiene cuenta.

\textbf{Flujo Principal:}
\begin{enumerate}
    \item El usuario ingresa sus datos
    \item El sistema valida los datos
    \item El sistema crea la cuenta
\end{enumerate}
